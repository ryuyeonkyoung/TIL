백트래킹
static void combination(int[] arr, boolean[] visited, int start, int n, int r) {
    if (r == 0) {
        print(arr, visited, n);
        return;
    }
    
    for (int i = start; i < n; i++) {
        visited[i] = true;                  // 현재 요소를 선택
        combination(arr, visited, i + 1, n, r - 1); // 다음 요소를 선택하기 위해 재귀 호출
        visited[i] = false;                 // 선택한 요소를 다시 방문하지 않도록 설정
    }
}

public static void permute(int[] arr, int[] out, boolean[] visited, int dept) {
	int n = arr.length;
	if (dept == out.length) {
		// out 배열 출력
	} else {
		for (int i = dept; i < arr.length; i++) {
			if (!visited[i]) {
				visited[i] = true;
				out[dept] = arr[i];
				// 다음 요소에 대해 재귀적으로 순열 생성
				permute(arr, out, visited, dept + 1);
				visited[i] = false; // 백트래킹
			}
		}
	}
}
